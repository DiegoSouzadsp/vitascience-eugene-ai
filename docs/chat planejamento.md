# chunking_strategy.py
def chunk_eugene_book(text):
    """Estratégia específica para o livro do Eugene"""
    
    # 1. Dividir por seções do livro
    sections = identify_book_sections(text)
    
    # 2. Chunk focado nos 5 níveis de consciência
    consciousness_chunks = extract_consciousness_levels(sections)
    
    # 3. Técnicas de copywriting específicas
    technique_chunks = extract_techniques(sections)
    
    # 4. Exemplos práticos do livro
    example_chunks = extract_examples(sections)
    
    return consciousness_chunks + technique_chunks + example_chunks

# retrieval_system.py
class EugeneRAG:
    def __init__(self):
        self.vector_db = init_vector_db()
        self.embedder = OpenAIEmbeddings()
    
    def search_relevant_concepts(self, vsl_text, consciousness_level=None):
        """Busca conceitos específicos do Eugene para a análise"""
        
        # Busca baseada no conteúdo da VSL
        content_results = self.vector_db.similarity_search(
            vsl_text, 
            k=3,
            filter={"type": "technique"}
        )
        
        # Busca específica para o nível de consciência
        if consciousness_level:
            consciousness_results = self.vector_db.similarity_search(
                f"nível {consciousness_level} consciência",
                k=2,
                filter={"type": "consciousness"}
            )
        
        return content_results + consciousness_results

# research_processor.py
class EugeneResearchProcessor:
    def __init__(self):
        self.validated_content = []
        self.confidence_threshold = 7
    
    def process_chatgpt_research(self, research_data):
        """Processa pesquisa do ChatGPT e estrutura para RAG"""
        
        validated_items = []
        for item in research_data:
            if item.confidence >= self.confidence_threshold:
                structured_item = {
                    'content': item.text,
                    'category': self.categorize_content(item),
                    'source_type': 'research_validated',
                    'confidence': item.confidence,
                    'metadata': {
                        'topic': item.topic,
                        'application': item.practical_use,
                        'examples': item.examples
                    }
                }
                validated_items.append(structured_item)
        
        return validated_items
